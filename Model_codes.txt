def clean_text(text):
  text=str(text).lower()
  text=re.sub(r'http\S+|www\S+',' ',text)
  text=re.sub(r"[^a-z\s]",' ',text)
  text=re.sub(r"\s+",' ',text)
  text=text.strip()
  return text

X_train_clean=X_train.apply(clean_text)
X_test_clean=X_test.apply(clean_text)

pipeline=Pipeline( 
    [('tfidf',TfidfVectorizer()),

)
